# Core runtime dependencies for the SoCal wildfire / grid co-simulation.
#
# This file is the COMPLETE install:
#     pip install -r requirements.txt
#
# It used to be only half of it -- pandapower had to be forced in afterwards
# with `pip install --no-deps -r requirements-nodeps.txt`, because palaestrAI
# pinned pandas~=2.1.4 while pandapower >= 3.2 declares pandas~=2.3, which pip
# rightly reported as ResolutionImpossible. That is fixed at the source:
# palaestrAI now allows pandas~=2.3, so pandapower installs through the normal
# resolver with its own dependencies and requirements-nodeps.txt is gone.
#
# Version selection rationale (Aug 2026):
# - pandas 2.3 is the ceiling, not a choice: pandapower 3.4.0 declares
#   pandas~=2.3, and pandas 3.x is not yet supported by pandapower.
# - numpy stays on 1.26.4 because palaestrAI pins numpy~=1.26.4. Nothing here
#   needs numpy 2: pandapower 3.4.0 accepts numpy>=1.26,<2.4 and pandas 2.3
#   needs only numpy>=1.26 on py3.12.
#   Moving to pandapower 3.5.x WOULD force numpy 2, because it requires
#   scipy~=1.18 on py3.12 and scipy 1.18 requires numpy>=2.0. That is a
#   separate migration; do not start it by bumping this file.
# - pandapower must be >= 3.2 for the stored grid JSONs (see the note on the
#   pandapower line below). tests/test_pandapower_compat.py enforces this.
#
# Power-flow results were verified bit-identical between pandas 2.1.4 and
# 2.3.3 across five reference grids (all result tables hashed at full float64
# precision), so this bump does not move any published number.
#
# Do not bump numpy, pandas, pandapower, or simbench without re-validating BOTH
# the test suite and a real MIDAS run -- an incompatible pandapower fails
# completely silently and produces plausible-looking all-zero results.
numpy==1.26.4
pandas==2.3.3
scipy>=1.11,<2
networkx>=3.4
matplotlib>=3.7

# pandapower >= 3.2 is required, not merely preferred. Our stored grid JSONs
# (socal_grid/socal_grid.json, midas_socal/*.json) use the split ZIP-load
# columns const_z_p_percent / const_i_p_percent, introduced in 3.2. On
# pandapower 3.1.x, runpp() looks for the old combined const_z_percent and
# raises KeyError; socal_grid.dispatch_and_run treats that as a non-convergent
# step, so a too-old pandapower does NOT crash the run -- it completes, exits 0,
# and silently produces an experiment in which the power flow never solved, no
# load was ever shed, and every grid KPI is exactly zero.
#
# pandapower now brings its own dependencies (packaging, tqdm, deepdiff,
# geojson, typing_extensions, pandera, ...) through the resolver. They used to
# be duplicated here by hand because of the --no-deps install; that hand-
# maintained mirror is deliberately removed -- it only drifted.
pandapower==3.4.0

# The remaining MIDAS simulator modules named in midas_socal/socal_midas.yml
# (`modules: [store, timesim, powergrid, weather, powerseries]`). MIDAS resolves
# each entry to a `midas_<name>` import at scenario-configure time, so a missing
# one is not caught until the run starts:
#     ModuleNotFoundError: No module named 'midas_store'
# which is exactly how the midas_smoke CI job failed. Only midas-powergrid was
# listed before, because the test suite imports it directly while the other four
# are reached solely through the YAML scenario.
# midas-mosaik[arl] covers store/timesim/weather but NOT powerseries, so these
# are pinned individually to match the versions the scenario is developed on.
midas-store==2.1.0
midas-timesim==2.1.1
midas-weather==2.1.0
midas-powerseries==2.1.0

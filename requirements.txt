#

numpy==1.26.4
pandas==2.1.4
scipy>=1.11,<2
pandapower==3.4.0
networkx>=3.0

#   python -m pip install --no-deps midas-powergrid==2.1.0 simbench==1.6.2 natsort joblib
# (simbench/natsort/joblib are midas-powergrid's runtime deps that are safe to add.)

# Core runtime dependencies for the SoCal wildfire / grid co-simulation.
#
# !! This file is only HALF the install. Always follow it with:
#        pip install --no-deps -r requirements-nodeps.txt
#    which brings in pandapower. See requirements-nodeps.txt for why it
#    cannot go through pip's resolver.
#
# Version selection rationale (Aug 2026):
# - palaestrAI 3.5.11 hard-pins numpy~=1.26.4 and pandas~=2.1.4, so numpy and
#   pandas are frozen here to match.
# - pandapower must be >= 3.2 for the stored grid JSONs, but its own declared
#   pins conflict with the two above, so it lives in requirements-nodeps.txt.
# - simbench 1.6.2 requires pandapower>=3.4.0 and would drag pandas 2.3 back
#   in through the resolver; 1.6.1 works and is what we test against.
#
# Do not bump numpy, pandas, pandapower, or simbench without re-validating
# BOTH the test suite and a real MIDAS run -- an incompatible pandapower fails
# completely silently and produces plausible-looking all-zero results.
numpy==1.26.4
pandas==2.1.4
scipy>=1.11,<2
# pandapower is intentionally NOT listed here -- see requirements-nodeps.txt.
